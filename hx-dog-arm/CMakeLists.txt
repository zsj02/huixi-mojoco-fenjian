cmake_minimum_required(VERSION 3.16)
project(hx_grid_environment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(interaction_state_test tests/interaction_state.cc)
target_include_directories(interaction_state_test PRIVATE src/hx_grid_sim/include)
add_test(NAME interaction_state COMMAND interaction_state_test)
option(BUILD_GAZEBO_PLUGIN "Build Gazebo Harmonic interaction plugin" ON)
if(BUILD_GAZEBO_PLUGIN)
  find_package(gz-sim8 REQUIRED)
  find_package(gz-plugin2 REQUIRED COMPONENTS register)
  find_package(gz-transport13 REQUIRED)
  add_library(hx_grid_interactions SHARED src/hx_grid_sim/src/GridInteractions.cc)
  target_include_directories(hx_grid_interactions PRIVATE src/hx_grid_sim/include)
  target_link_libraries(hx_grid_interactions gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13)
endif()
